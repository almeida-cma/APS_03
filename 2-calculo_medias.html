def button_calcular_click(self, **event_args):
    """
    ========================================
    EVENTO: click do botão CALCULAR
    DISPARADO QUANDO: Usuário clica no botão
    ========================================
    """

    # =========================
    # ENTRADA DE DADOS
    # Captura os valores digitados nos campos
    # =========================

    nota1_texto = self.text_box_nota_1.text
    nota2_texto = self.text_box_nota_2.text
    nota3_texto = self.text_box_nota_3.text

    # =========================
    # VALIDAÇÃO DE CAMPOS VAZIOS
    # Verifica se o usuário preencheu todas as notas
    # =========================

    if not nota1_texto or not nota2_texto or not nota3_texto:
        self.label_resultado.text = "⚠️ Preencha todas as notas!"
        self.label_resultado.foreground = "#e74c3c"  # Vermelho
        return  # Interrompe a execução

    # =========================
    # CONVERSÃO PARA NÚMERO
    # Tenta converter texto para número decimal
    # =========================

    try:
        nota1 = float(nota1_texto)
        nota2 = float(nota2_texto)
        nota3 = float(nota3_texto)

    except (ValueError, TypeError):
        # Se não conseguir converter (letras, símbolos ou valores vazios)
        self.label_resultado.text = "❌ Digite apenas números válidos!"
        self.label_resultado.foreground = "#e74c3c"
        return

    # =========================
    # VALIDAÇÃO DO INTERVALO (0 a 10)
    # As notas devem estar entre 0 e 10
    # =========================

    if nota1 < 0 or nota1 > 10:
        self.label_resultado.text = "❌ Nota 1 inválida! Use valores de 0 a 10"
        self.label_resultado.foreground = "#e74c3c"
        return

    if nota2 < 0 or nota2 > 10:
        self.label_resultado.text = "❌ Nota 2 inválida! Use valores de 0 a 10"
        self.label_resultado.foreground = "#e74c3c"
        return

    if nota3 < 0 or nota3 > 10:
        self.label_resultado.text = "❌ Nota 3 inválida! Use valores de 0 a 10"
        self.label_resultado.foreground = "#e74c3c"
        return

    # =========================
    # PROCESSAMENTO
    # Calcula a média aritmética
    # =========================

    media = (nota1 + nota2 + nota3) / 3

    # =========================
    # REGRA DE NEGÓCIO
    # Define a situação baseada na média
    # =========================

    if media >= 7:
        situacao = "APROVADO"
        cor = "#27ae60"      # Verde
        emoji = "✅"

    elif media >= 5:
        situacao = "RECUPERAÇÃO"
        cor = "#f39c12"      # Laranja
        emoji = "⚠️"

    else:
        situacao = "REPROVADO"
        cor = "#e74c3c"      # Vermelho
        emoji = "❌"

    # =========================
    # SAÍDA
    # Exibe o resultado formatado na tela
    # =========================

    self.label_resultado.text = f"{emoji} Média: {media:.1f} - {situacao} {emoji}"
    self.label_resultado.foreground = cor
    self.label_resultado.font_size = 18
    self.label_resultado.bold = True

    # Debug no console do ANVIL
    print(f"Média calculada: {media:.2f} - Situação: {situacao}")
📖 ETAPA 8 — EXPLICAÇÃO DETALHADA DO CÓDIGO Entenda a lógica linha por linha

📥 ENTRADA

Usamos self.[id_do_componente].text para puxar os dados. No Anvil, self é a tela atual, e o ID é o nome que você deu ao componente no painel Properties.

🛡️ VALIDAÇÃO SEGURA

O uso de if not nota_texto: é a forma "Pythônica" ideal para verificar nulos e strings vazias. Se o usuário esquecer um campo, o return cancela o cálculo na hora!

⚠️ TRATAMENTO DE ERROS

O bloco try/except (ValueError, TypeError) é vital. Ele tenta converter texto para Decimal. Se o usuário digitar "ABC" (Value) ou enviar objeto nulo (Type), o except entra em ação e impede que o app trave ("quebre").

📏 LIMITE DE NOTAS

Usando operadores lógicos if nota < 0 or nota > 10:, garantimos que a matemática de notas não seja burlada pelo usuário.

🧮 PROCESSAMENTO & NEGÓCIO

Variáveis temporárias guardam a matemática media = (n1+n2+n3)/3. Depois, o famoso if / elif / else decide dinamicamente as strings de texto e cores hexadecimais.

📤 SAÍDA VISUAL

Manipulamos o label_resultado injetando a F-String (para formatar texto e variáveis juntos). Usamos .foreground no Python para pintar o texto do frontend em tempo real!
