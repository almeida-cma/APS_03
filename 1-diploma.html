def button_verificar_click(self, **event_args):
    
    # =========================
    # ENTRADA DE DADOS
    # O sistema PEGA o que o usuário digitou
    # =========================
    
    nota_texto = self.text_box_nota.text
    frequencia_texto = self.text_box_freq.text
    
    
    # =========================
    # VALIDAÇÃO 1 (Campos vazios)
    # O sistema VERIFICA se faltou alguma informação
    # =========================
    
    if nota_texto == "" or frequencia_texto == "":
        self.label_resultado.text = "⚠️ Preencha todos os campos!"
        self.label_resultado.role = None
        return
    
    
    # =========================
    # CONVERSÃO PARA NÚMERO
    # O sistema TRANSFORMA texto em número
    # =========================
    
    try:
        nota = float(nota_texto)
        frequencia = float(frequencia_texto)
    except:
        self.label_resultado.text = "❌ Digite apenas números!"
        self.label_resultado.role = None
        return
    
    
    # =========================
    # VALIDAÇÃO 2 (Limites Permitidos)
    # VERIFICA limites (0-10, 0-100)
    # =========================
    
    if nota < 0 or nota > 10:
        self.label_resultado.text = "❌ Nota inválida! Digite um valor entre 0 e 10."
        self.label_resultado.role = None
        return
    
    if frequencia < 0 or frequencia > 100:
        self.label_resultado.text = "❌ Frequência inválida! Digite um valor entre 0 e 100."
        self.label_resultado.role = None
        return
    
    
    # =========================
    # REGRA DE NEGÓCIO (O Coração do Sistema)
    # AND = ambas condições devem ser verdadeiras
    # =========================
    
    if nota >= 7 and frequencia >= 75:
        situacao = "✅ DIPLOMA CONCEDIDO!"
        cor = "success"
        
    else:
        cor = "danger"
        
        # Mensagem adicional explicando o motivo exato da reprovação
        if nota < 7 and frequencia < 75:
            motivo = "Nota abaixo de 7 E frequência abaixo de 75%"
        elif nota < 7:
            motivo = "Nota abaixo de 7"
        else:
            motivo = "Frequência abaixo de 75%"
        
        situacao = f"❌ NÃO ATENDE AOS REQUISITOS\n({motivo})"
    
    
    # =========================
    # SAÍDA
    # O sistema MOSTRA o resultado para o usuário
    # =========================
    
    self.label_resultado.text = situacao
    self.label_resultado.role = cor
📖 ETAPA 8 — EXPLICANDO O CÓDIGO E TESTES Compreensão e Validação

📥 Entrada e Conversão

Textos da tela entram como String. Usamos float() para que o Python entenda que são números que podem ter casas decimais.

🧮 Lógica Condicional (O IF/ELSE)

O if ... else permite que o sistema tome uma DECISÃO baseada nas regras impostas pela faculdade, automatizando o papel da secretaria.

🎨 A Propriedade `.role`

O Anvil permite usar temas (Roles). Ao definirmos .role = "success" via código, o Anvil pinta o label de verde automaticamente. "danger" o pinta de vermelho.
